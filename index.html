  unknown: { label: 'New — no comparison yet',     pillBg: 'bg-slate-100',  pillText: 'text-slate-600',   dot: 'bg-slate-400',   stroke: '#94a3b8' },

function StatusPill({ severity, qualifier }) {

  const s = SEVERITY[severity];

  return (

    <span className={`inline-flex items-center gap-1.5 px-2.5 py-1 rounded-full text-[11px] font-medium ${s.pillBg} ${s.pillText} whitespace-nowrap`}>

      <span className={`w-1.5 h-1.5 rounded-full ${s.dot}`}></span>

      {s.label}

      {qualifier && <span className="font-normal opacity-75"> · {qualifier}</span>}

    </span>

  );

}


// Maps a value to a 0-100% position on a 5-zone severity bar

// (15% severe-low | 10% mild-low | 50% in-range | 10% mild-high | 15% severe-high)

function getDotPosition(value, low, high) {

  const range = high - low;

  if (value <= low) {

    const u = (low - value) / range;

    if (u < 0.25) return 25 - (u / 0.25) * 10;

    if (u < 1.0)  return 15 - ((u - 0.25) / 0.75) * 13;

    return 2;

  }

  if (value <= high) return 25 + ((value - low) / range) * 50;

  const o = (value - high) / range;

  if (o < 0.25) return 75 + (o / 0.25) * 10;

  if (o < 1.0)  return 85 + ((o - 0.25) / 0.75) * 13;

  return 98;

}


function RangeStrip({ value, low, high, severity }) {

  const pct = getDotPosition(value, low, high);

  return (

    <div className="relative pt-3 pb-5 mt-3">

      <div className="flex h-2 rounded-full overflow-hidden">

        <div className="bg-red-100"     style={{ width: '15%' }}></div>

        <div className="bg-amber-100"   style={{ width: '10%' }}></div>

        <div className="bg-emerald-100" style={{ width: '50%' }}></div>

        <div className="bg-amber-100"   style={{ width: '10%' }}></div>

        <div className="bg-red-100"     style={{ width: '15%' }}></div>

      </div>

      <div className="absolute top-[10px] -translate-x-1/2" style={{ left: `${pct}%` }}>

        <div className={`w-3.5 h-3.5 rounded-full border-[2.5px] border-white shadow-md ${SEVERITY[severity].dot}`}></div>

      </div>

      <div className="absolute bottom-0 left-0 right-0 text-[10px] text-slate-400 font-medium tabular h-3">

        <span className="absolute -translate-x-1/2" style={{ left: '25%' }}>{low}</span>

        <span className="absolute -translate-x-1/2" style={{ left: '75%' }}>{high}</span>

      </div>

    </div>

  );

}


function Sparkline({ points, stroke }) {

  if (!points || points.length < 2) return null;

  const min = Math.min(...points);

  const max = Math.max(...points);

  const r = max - min || 1;

  const w = 72, h = 22;

  const path = points.map((p, i) => {

    const x = (i / (points.length - 1)) * w;

    const y = h - ((p - min) / r) * h;

    return `${i === 0 ? 'M' : 'L'} ${x.toFixed(1)} ${y.toFixed(1)}`;

  }).join(' ');

  const lastX = w;

  const lastY = h - ((points[points.length - 1] - min) / r) * h;

  return (

    <svg width={w} height={h} className="overflow-visible flex-shrink-0">

      <path d={path} fill="none" stroke={stroke} strokeWidth="1.75" strokeLinecap="round" strokeLinejoin="round" />

      <circle cx={lastX} cy={lastY} r="2.25" fill={stroke} />

    </svg>

  );

}


// Arrow direction always matches the actual numeric change.

// Color reflects "toward-healthy" (good) vs "away-from-healthy" (bad).

function TrendRow({ delta, unit, sinceDate, sparklinePoints, trendDirection, trendSeverity }) {

  const arrow = trendDirection === 'up' ? '↑' : trendDirection === 'down' ? '↓' : '→';

  const colorClass =

    trendSeverity === 'good' ? 'text-emerald-600' :

    trendSeverity === 'bad'  ? 'text-orange-600'  :

                                'text-slate-500';

  const stroke =

    trendSeverity === 'good' ? '#10b981' :

    trendSeverity === 'bad'  ? '#ea580c' :

                                '#94a3b8';

  return (

    <div className="flex items-center justify-between gap-3 py-3 border-t border-slate-100 mt-2">

      <div className="flex items-baseline gap-1.5 text-xs">

        <span className={`font-semibold ${colorClass} text-[13px] tabular`}>{arrow} {Math.abs(delta)} {unit}</span>

        <span className="text-slate-400">since {sinceDate}</span>

      </div>

      <Sparkline points={sparklinePoints} stroke={stroke} />

    </div>

  );

}
